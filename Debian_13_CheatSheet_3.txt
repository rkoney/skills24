CA (OpenSSL)
mkdir -p /root/ca/{root,sub} /opt/grading/ca
cd /root/ca/root

# Root CA
openssl genrsa -out root.key 4096
openssl req -x509 -new -nodes -key root.key -sha256 -days 3650 \
  -subj "/CN=ClearSky Root CA/C=CH/O=ClearSky" -out root.pem
cp root.pem /opt/grading/ca/ca.pem

# Sub-CA
cd /root/ca/sub
openssl genrsa -out sub.key 4096
openssl req -new -key sub.key -subj "/CN=ClearSky Sub CA/C=CH/O=ClearSky" -out sub.csr
openssl x509 -req -in sub.csr -CA ../root/root.pem -CAkey ../root/root.key \
  -CAcreateserial -days 1825 -sha256 \
  -extfile <(printf "basicConstraints=CA:TRUE") -out sub.pem
cp sub.pem /opt/grading/ca/services.pem

# Web certificate
openssl genrsa -out web.key 2048
openssl req -new -key web.key -subj "/CN=www.dmz.redseason.org" -out web.csr
openssl x509 -req -in web.csr -CA sub.pem -CAkey sub.key -CAcreateserial -days 825 \
  -extfile <(printf "subjectAltName=DNS:www.dmz.redseason.org") -out web.pem
cat web.pem sub.pem root.pem > /opt/grading/ca/web.pem

# Mail certificate
openssl genrsa -out mail.key 2048
openssl req -new -key mail.key -subj "/CN=mail.dmz.redseason.org" -out mail.csr
openssl x509 -req -in mail.csr -CA sub.pem -CAkey sub.key -CAcreateserial -days 825 \
  -extfile <(printf "subjectAltName=DNS:mail.dmz.redseason.org") -out mail.pem
cat mail.pem sub.pem root.pem > /opt/grading/ca/mail.pem
Copy web.key / mail.key to the servers that need them via scp (root SSH key is pre-provisioned on all hosts).

/etc/samba/smb.conf:
[global]
   workgroup = WORKGROUP
   security = user
   map to guest = Bad User

[public]
   path = /srv/public
   guest ok = yes
   read only = yes
   valid users = jamie
   write list = jamie

[internal]
   path = /srv/internal
   guest ok = no
   read only = no
   valid users = jamie
/public: everyone (including guests) can read; only jamie (logged-in) can write, via write list. /internal: guest ok = no blocks guests entirely; only authenticated valid users get read/write.

nftables
apt install nftables -y
/etc/nftables.conf (core rules):
#!/usr/sbin/nft -f
flush ruleset

define WAN = eth0
define INT = eth1
define DMZ = eth2
define VPN = wg0

table inet filter {
  chain forward {
    type filter hook forward priority 0; policy drop;

    ct state established,related accept

    # INT/DMZ -> internet
    iifname { $INT, $DMZ } oifname $WAN accept
    # INT -> DMZ
    iifname $INT oifname $DMZ accept
    # VPN -> INT/DMZ
    iifname $VPN oifname { $INT, $DMZ } accept
    # mail server -> LDAP on int-srv01
    ip saddr 10.1.20.10 ip daddr 10.1.10.10 tcp dport 389 accept

    # port forwards (destined for DMZ HA proxy)
    iifname $WAN ip daddr 10.1.20.20 tcp dport { 80, 443 } accept
    iifname $WAN ip daddr 10.1.20.20 udp dport 53 accept
    iifname $WAN ip daddr 10.1.20.20 tcp dport 53 accept
  }
  chain input {
    type filter hook input priority 0; policy accept;
  }
}

table ip nat {
  chain postrouting {
    type nat hook postrouting priority 100;
    oifname $WAN masquerade
  }
  chain prerouting {
    type nat hook prerouting priority -100;
    iifname $WAN tcp dport 80 dnat to 10.1.20.20:80
    iifname $WAN tcp dport 443 dnat to 10.1.20.20:443
    iifname $WAN udp dport 53 dnat to 10.1.20.20:53
    iifname $WAN tcp dport 53 dnat to 10.1.20.20:53
  }
}

table ip6 filter {
  chain forward {
    type filter hook forward priority 0; policy drop;
    ct state established,related accept
    iifname { $INT, $DMZ } oifname $WAN accept
    iifname $INT oifname $DMZ accept
    iifname $VPN oifname { $INT, $DMZ } accept
    iifname $WAN ip6 daddr 2001:db8:1001:20::20 tcp dport { 80, 443 } accept
  }
}
systemctl enable --now nftables

mail.dmz.redseason.org
Postfix (SMTP) + Dovecot (IMAP), authenticated against LDAP on int-srv01, TLS secured with the CA-issued mail.pem.

Copy the mail certificate/key generated in Part 1 onto this host:

Postfix — /etc/postfix/main.cf
# Identity
myhostname = mail.dmz.redseason.org
mydomain = dmz.redseason.org
myorigin = $mydomain
mydestination = $myhostname, dmz.redseason.org, localhost
inet_interfaces = all
inet_protocols = ipv4, ipv6

# TLS (server)
smtpd_tls_cert_file = /etc/ssl/certs/mail.pem
smtpd_tls_key_file = /etc/ssl/private/mail.key
smtpd_use_tls = yes
smtpd_tls_security_level = may
smtpd_tls_auth_only = yes

# TLS (outbound / client)
smtp_tls_security_level = may

# SASL auth via Dovecot socket (for authenticated relay/send)
smtpd_sasl_type = dovecot
smtpd_sasl_path = private/auth
smtpd_sasl_auth_enable = yes
smtpd_recipient_restrictions =
    permit_sasl_authenticated,
    permit_mynetworks,
    reject_unauth_destination

# Mailbox delivery (Dovecot LMTP)
virtual_transport = lmtp:unix:private/dovecot-lmtp

# Aliases (used by the echo auto-responder)
alias_maps = hash:/etc/aliases
alias_database = hash:/etc/aliases
/etc/postfix/master.cf — ensure submission (587) is enabled for client sending, uncomment/add:
submission inet n       -       y       -       -       smtpd
  -o syslog_name=postfix/submission
  -o smtpd_tls_security_level=encrypt
  -o smtpd_sasl_auth_enable=yes
  -o smtpd_recipient_restrictions=permit_sasl_authenticated,reject
systemctl restart postfix

Dovecot — IMAP + LDAP auth + LMTP delivery
/etc/dovecot/dovecot.conf (top-level, ensure protocols include lmtp):

mail_privileged_group = mail
/etc/dovecot/conf.d/10-auth.conf:

auth_mechanisms = plain login
!include auth-ldap.conf.ext
/etc/dovecot/conf.d/auth-ldap.conf.ext:
passdb {
  driver = ldap
  args = /etc/dovecot/dovecot-ldap.conf.ext
}
userdb {
  driver = ldap
  args = /etc/dovecot/dovecot-ldap.conf.ext
}
/etc/dovecot/dovecot-ldap.conf.ext — binds to int-srv01, maps LDAP mail attribute to the mailbox user:
hosts = 10.1.10.10
base = ou=Employees,dc=int,dc=redseason,dc=org
auth_bind = yes
auth_bind_userdn = uid=%n,ou=Employees,dc=int,dc=redseason,dc=org

user_attrs = mail=user,mail=home=/var/mail/vhosts/%d/%n
user_filter = (uid=%n)
pass_filter = (uid=%n)
pass_attrs = uid=user
user_attrs maps LDAP's mail attribute (e.g. jamie.oliver@dmz.redseason.org) directly to Dovecot's mailbox user, satisfying "the mail attribute of the LDAP user should be used as the mail address."
/etc/dovecot/conf.d/10-ssl.conf — TLS required, using the CA-issued certificate:
ssl = required
ssl_cert = </etc/ssl/certs/mail.pem
ssl_key = </etc/ssl/private/mail.key
/etc/dovecot/conf.d/10-master.conf — expose the SASL auth socket to Postfix and enable LMTP delivery:
service auth {
  unix_listener /var/spool/postfix/private/auth {
    mode = 0666
    user = postfix
    group = postfix
  }
}

service lmtp {
  unix_listener /var/spool/postfix/private/dovecot-lmtp {
    mode = 0600
    user = postfix
    group = postfix
  }
}
/etc/dovecot/conf.d/20-imap.conf — confirm standard ports are active (143 plain, 993 IMAPS):
protocol imap {
  mail_max_userip_connections = 20
}

ha-prx01 / ha-prx02
High-availability reverse proxy (HAProxy + keepalived) and primary/secondary DNS (BIND9).
apt install haproxy keepalived bind9 -y

keepalived (VRRP)
/etc/keepalived/keepalived.conf — ha-prx01 = MASTER (higher priority), ha-prx02 = BACKUP:
vrrp_instance VI_1 {
    state MASTER
    interface eth0
    virtual_router_id 51
    priority 150
    virtual_ipaddress {
        10.1.20.20/24
        2001:db8:1001:20::20/64
    }
    track_script { chk_haproxy }
}
vrrp_script chk_haproxy {
    script "pidof haproxy"
    interval 2
}
On ha-prx02, use state BACKUP and a lower priority (e.g. 100) — same virtual_router_id.
HAProxy
/etc/haproxy/haproxy.cfg:
frontend http_front
    bind *:80
    redirect scheme https code 301

frontend https_front
    bind *:443 ssl crt /etc/ssl/private/web-combined.pem
    default_backend web_back
    http-response add-header via-proxy ha-prx01.dmz.redseason.org

backend web_back
    balance roundrobin
    server web01 10.1.20.31:80 check
    server web02 10.1.20.32:80 check
Combine web.key + the web.pem chain into a single PEM (web-combined.pem) for HAProxy's bind ... ssl crt. Replace via-proxy hostname value with ha-prx02.dmz.redseason.org on the second node.
Reverse Proxy — Nginx alternative
If you'd rather use Nginx instead of HAProxy for the reverse proxy role, this achieves the same requirements (HTTP→HTTPS redirect, load balancing across web01/web02, TLS termination, via-proxy header, failover to the other web server).
apt install nginx -y
Place the certificate chain on the host:
cp web.key /etc/ssl/private/web.key
cp web.pem /etc/ssl/certs/web.pem   # already contains web+sub+root chain from Part 1
chmod 600 /etc/ssl/private/web.key
/etc/nginx/sites-available/dmz-proxy:
upstream web_backend {
    server 10.1.20.31:80 max_fails=2 fail_timeout=5s;
    server 10.1.20.32:80 max_fails=2 fail_timeout=5s;
}

# Plain HTTP -> redirect to HTTPS
server {
    listen 80;
    listen [::]:80;
    server_name www.dmz.redseason.org;
    return 301 https://$host$request_uri;
}

# HTTPS termination + reverse proxy
server {
    listen 443 ssl;
    listen [::]:443 ssl;
    server_name www.dmz.redseason.org;

    ssl_certificate     /etc/ssl/certs/web.pem;
    ssl_certificate_key /etc/ssl/private/web.key;
    ssl_protocols       TLSv1.2 TLSv1.3;

    location / {
        proxy_pass http://web_backend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # troubleshooting header identifying which proxy node served the request
        add_header via-proxy "ha-prx01.dmz.redseason.org" always;

        # automatic failover if one web server is down
        proxy_next_upstream error timeout http_502 http_503 http_504;
    }
}
ln -s /etc/nginx/sites-available/dmz-proxy /etc/nginx/sites-enabled/
rm -f /etc/nginx/sites-enabled/default
nginx -t
systemctl restart nginx
On ha-prx02, deploy the identical config but change the add_header via-proxy value to ha-prx02.dmz.redseason.org. proxy_next_upstream plus the upstream block's max_fails/fail_timeout give Nginx equivalent failover behavior to HAProxy's health checks — if web01 stops responding, requests are automatically retried against web02. The virtual-IP takeover on proxy failure (10.1.20.20) is still handled by keepalived regardless of whether you run HAProxy or Nginx as the actual proxy process — just point keepalived's chk_haproxy script at pidof nginx instead:
vrrp_script chk_haproxy {
    script "pidof nginx"
    interval 2
}
